    @Test
    public void t11_getByIdEndpointReturnsEntity() {
        try {
            Class<?> controllerClass = Class.forName("com.example.demo.controller.GraphController");
            Class<?> serviceClass = Class.forName("com.example.demo.service.GraphOrchestratorService");
            Class<?> entityClass = Class.forName("com.example.demo.entity.KnowledgeGraph");
            Class<?> repoClass = Class.forName("com.example.demo.repository.KnowledgeGraphRepository");

            Object graph = entityClass.getDeclaredConstructor().newInstance();
            Method setId = entityClass.getMethod("setId", Long.class);
            setId.invoke(graph, 1L);

            // Repository: findById(1L) returns the graph
            Object repoProxy = Proxy.newProxyInstance(
                    repoClass.getClassLoader(),
                    new Class<?>[]{repoClass},
                    (proxy, method, args) -> {
                        if (method.getName().equals("findById") && args != null && args.length > 0 && args[0].equals(1L)) {
                            return Optional.of(graph);
                        }
                        return Optional.empty();
                    }
            );

            Class<?> eventPublisherClass = Class.forName("org.springframework.context.ApplicationEventPublisher");
            Object eventProxy = Proxy.newProxyInstance(
                    eventPublisherClass.getClassLoader(),
                    new Class<?>[]{eventPublisherClass},
                    (proxy, method, args) -> null
            );
            Constructor<?> svcCtor = serviceClass.getDeclaredConstructor(repoClass, eventPublisherClass);
            svcCtor.setAccessible(true);
            Object service = svcCtor.newInstance(repoProxy, eventProxy);

            Object controller = createController(controllerClass, service);

            // Find GET method with {id} – should be @GetMapping with value "/{id}"
            Method getByIdMethod = null;
            for (Method m : controllerClass.getDeclaredMethods()) {
                if (m.isAnnotationPresent(getAnnotationClass("org.springframework.web.bind.annotation.GetMapping"))) {
                    Annotation ann = m.getAnnotation(getAnnotationClass("org.springframework.web.bind.annotation.GetMapping"));
                    String[] paths = (String[]) ann.getClass().getMethod("value").invoke(ann);
                    if (paths.length > 0 && paths[0].equals("/{id}")) {
                        getByIdMethod = m;
                        break;
                    }
                }
            }
            Assert.assertNotNull(getByIdMethod, "No method found for GET /{id}");
            getByIdMethod.setAccessible(true);

            Object result = getByIdMethod.invoke(controller, 1L);
            Assert.assertNotNull(result);
            if (result instanceof ResponseEntity) {
                ResponseEntity<?> re = (ResponseEntity<?>) result;
                Assert.assertEquals(re.getStatusCode(), HttpStatus.OK);
                Object body = re.getBody();
                Assert.assertNotNull(body);
                Method getId = body.getClass().getMethod("getId");
                Assert.assertEquals(getId.invoke(body), 1L);
            } else {
                Assert.fail("Expected ResponseEntity");
            }
        } catch (ClassNotFoundException e) {
            Assert.fail("Required class not found: " + e.getMessage());
        } catch (Exception e) {
            Assert.fail("GET by ID test failed: " + e.getMessage());
        }
    }
